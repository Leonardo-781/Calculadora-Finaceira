#include <stdio.h>
#include <math.h>

int main() {
    // Declaração de variáveis
    float I, i, n, fc, vpl = 0, temp, VF = 0;
    int vres, periodoVR;

    // Entrada de dados
    printf("\n\tOpcao Selecionada: VPL - Valor Presente Liquido.\n\n");
    printf("Insira o valor do Investimento (I): ");
    scanf("%f", &I);

    printf("Insira a taxa (i): ");
    scanf("%f", &i);

    printf("Insira o Tempo (n): ");
    scanf("%f", &n);

    // Cálculo do VPL
    for (int j = 1; j <= n; j++) {
        printf("Insira o Fluxo de Caixa do %d^o periodo: ", j);
        scanf("%f", &fc);

        temp = fc / pow((1 + i), j);
        vpl += temp;
    }

    // Verificação de valor residual
    printf("\nTem valor residual?\n1-Sim\n2-Nao\nEscolha uma opcao: ");
    scanf("%d", &vres);

    switch (vres) {
        case 1:
            // Entrada de dados para valor residual
            printf("\nInsira o Valor Residual (VR): ");
            scanf("%f", &VF);
            printf("\nInsira o periodo do valor residual: ");
            scanf("%d", &periodoVR);

            // Ajuste do VPL com valor residual
            VF = VF / pow((i + 1), periodoVR);
            vpl = (vpl + VF) - I;

            break;

        case 2:
            // Ajuste do VPL sem valor residual
            vpl = (vpl + 0) - I;
            break;
    }

    // Saída de resultados
    printf("\n-----------------");
    printf("\nVPL = R$%.2f", vpl);
    printf("\n-----------------");

    

    return 0;
}
